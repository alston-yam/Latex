    \section{Problems}

    \begin{problem}
        The sum of the digits of the number $2010$ is $3$.

        What is the next larger number whose digit-sum is also $3$?
    \end{problem}

    \begin{problem}
        In the pattern below, what will the $78$th letter?

        \[A B B C C D A B B C C D A \cdots\]
    \end{problem}

    \begin{problem}
        If you triple Jen's age and subtract $16$, the result will be the same as when you double her age and add $8$. 

        How old is Jen?
    \end{problem}

    \begin{problem}
        This figure consists of two squares. Each side length is a whole number of centimetres. The combined areas of the squares is $100cm^2$. 

        What is the perimeter of the entire figure?

        \begin{center}
            \includegraphics[width=0.3\textwidth]{0201.jpeg}
        \end{center}
    \end{problem}

    \begin{problem}
        A pile of coins sits on a table. Sara takes half of the coins plus $4$ more. Then Nick takes $2$. Then Joe takes $2$ more than half of what is left. Finally Selena takes $5$. Four coins remain on the table.

        How many coins were on the table at the start?
    \end{problem}
