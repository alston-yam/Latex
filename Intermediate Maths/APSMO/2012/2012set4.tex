    \section{Problems}

    \begin{problem}
        Consecutive numbers are counting numbers that follow in order. For example: $3, 4, 5, 6$ are four consecutive numbers.

        Which four consecutive numbers have a sum equal to $70$?
    \end{problem}

    \begin{problem}
        $X * Y$ represents the fraction $\frac{(X + Y)}{(X - Y)}$.

        Express the number represented by $12 * (7 * 5)$ in its simplest form.
    \end{problem}

    \begin{problem}
        There are five Mondays in a certain month.

        If three of the five Mondays occur on even days, which day of the week is the $11$th day of the month?
    \end{problem}

    \begin{problem}
        In the division problem below, different letters represent different digits. $AB$ is an even $2$-digit number. $EEE$ is a $3$-digit number.

        Find the $2$-digit number $AB$.

        \begin{center}
            \includegraphics[width=0.25\textwidth]{0401.jpeg}
        \end{center}
    \end{problem}

    \begin{problem}
        A rectangular prism that is $4$cm by $6$cm by $8$cm is painted on all six faces.

        The prism is cut into cubes that measure $2$cm on each side.

        How many of these cubes have only one face painted?
    \end{problem}
